cmake_minimum_required(VERSION 3.16)
project(phone_orb_slam3)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORB_SLAM3_ROOT "" CACHE PATH "Absolute path to built upstream ORB_SLAM3")
if(NOT EXISTS "${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so")
  message(FATAL_ERROR "Build upstream ORB-SLAM3 first and pass -DORB_SLAM3_ROOT=/absolute/path")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
add_executable(phone_mono src/phone_mono.cpp)
ament_target_dependencies(phone_mono rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs cv_bridge)
target_include_directories(phone_mono PRIVATE ${ORB_SLAM3_ROOT} ${ORB_SLAM3_ROOT}/include ${ORB_SLAM3_ROOT}/include/CameraModels ${ORB_SLAM3_ROOT}/Thirdparty/Sophus ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
target_link_libraries(phone_mono ${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so ${OpenCV_LIBS} ${Pangolin_LIBRARIES} Eigen3::Eigen)
set_target_properties(phone_mono PROPERTIES INSTALL_RPATH "${ORB_SLAM3_ROOT}/lib;${ORB_SLAM3_ROOT}/Thirdparty/DBoW2/lib;${ORB_SLAM3_ROOT}/Thirdparty/g2o/lib")
install(TARGETS phone_mono DESTINATION lib/${PROJECT_NAME})
ament_package()
